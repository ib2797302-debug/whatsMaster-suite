#### `backend/controllers/addonController.js`
```javascript
const Addon = require('../models/Addon');
const TenantAddon = require('../models/TenantAddon');
const stripe = require('stripe')(process.env.STRIPE_SECRET_KEY);

exports.getAddons = async (req, res) => {
  const addons = await Addon.find({ isActive: true });
  res.json(addons);
};

exports.subscribeAddon = async (req, res) => {
  const { addonId, interval = 'monthly' } = req.body;
  const tenant = req.tenant;

  const addon = await Addon.findById(addonId);
  if (!addon) return res.status(404).json({ error: 'Addon not found' });

  const existing = await TenantAddon.findOne({ tenantId: tenant._id, addonId, status: 'active' });
  if (existing) return res.status(400).json({ error: 'Already subscribed' });

  let customerId = tenant.stripeCustomerId;
  if (!customerId) {
    const customer = await stripe.customers.create({
      email: tenant.name,
      metadata: { tenantId: tenant._id.toString() }
    });
    customerId = customer.id;
    tenant.stripeCustomerId = customerId;
    await tenant.save();
  }

  const priceId = interval === 'monthly' ? addon.stripePriceMonthlyId : addon.stripePriceYearlyId;
  if (!priceId) {
    return res.status(400).json({ error: 'No price configured for this addon' });
  }

  const subscription = await stripe.subscriptions.create({
    customer: customerId,
    items: [{ price: priceId }],
    metadata: { tenantId: tenant._id.toString(), addonId: addonId }
  });

  const tenantAddon = new TenantAddon({
    tenantId: tenant._id,
    addonId,
    stripeSubscriptionId: subscription.id,
    endDate: interval === 'monthly' ? new Date(Date.now() + 30*24*60*60*1000) : new Date(Date.now() + 365*24*60*60*1000)
  });
  await tenantAddon.save();

  res.status(201).json(tenantAddon);
};

exports.cancelAddon = async (req, res) => {
  const { addonId } = req.params;
  const tenantAddon = await TenantAddon.findOne({ tenantId: req.tenant._id, addonId, status: 'active' });
  if (!tenantAddon) return res.status(404).json({ error: 'Subscription not found' });

  await stripe.subscriptions.update(tenantAddon.stripeSubscriptionId, { cancel_at_period_end: true });
  tenantAddon.status = 'cancelled';
  await tenantAddon.save();

  res.json({ success: true });
};
```

```javascript
const Conversation = require('../models/Conversation');
const Message = require('../models/Message');
const { escapeRegex } = require('../utils/helpers');

exports.getConversations = async (req, res) => {
  try {
    const { search, status, limit = 20, cursor } = req.query;
    let filter = { tenantId: req.tenant._id };

    if (search) {
      const escaped = escapeRegex(search);
      filter.$or = [
        { 'contactId.name': new RegExp(escaped, 'i') },
        { whatsappId: new RegExp(escaped, 'i') }
      ];
    }
    if (status) filter.status = status;
    if (cursor) filter._id = { $lt: cursor };

    const conversations = await Conversation.find(filter)
      .populate('contactId')
      .sort({ _id: -1 })
      .limit(parseInt(limit))
      .lean()
      .select('_id contactId whatsappId lastMessageAt status assignedTo');

    const nextCursor = conversations.length === parseInt(limit) ? conversations[conversations.length-1]._id : null;

    res.json({ conversations, nextCursor });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};
```

#### `backend/middleware/permissions.js`
```javascript
const Role = require('../models/Role');

const hasPermission = (requiredPermission) => async (req, res, next) => {
  const user = req.user;
  if (!user) return res.status(401).json({ error: 'Unauthorized' });

  if (user.role === 'admin') return next();

  const role = await Role.findOne({ _id: user.roleId, tenantId: user.tenantId });
  if (!role) return res.status(403).json({ error: 'No role assigned' });

  if (role.permissions.includes(requiredPermission) || role.permissions.includes('*')) {
    return next();
  }
  return res.status(403).json({ error: 'Insufficient permissions' });
};

module.exports = { hasPermission };
```

```javascript
const AuditLog = require('../models/AuditLog');

const logAction = (action, resourceType, getResourceId = null, getOldData = null, getNewData = null) => {
  return async (req, res, next) => {
    let oldData, newData, resourceId;
    if (getOldData) oldData = await getOldData(req);
    if (getNewData) newData = getNewData(req);
    if (getResourceId) resourceId = getResourceId(req);

    res.on('finish', async () => {
      if (req.user && req.tenant && res.statusCode >= 200 && res.statusCode < 300) {
        await AuditLog.create({
          tenantId: req.tenant._id,
          userId: req.user._id,
          action,
          resourceType,
          resourceId,
          oldData,
          newData,
          ip: req.ip,
          userAgent: req.get('user-agent')
        });
      }
    });
    next();
  };
};

module.exports = { logAction };
```

#### `backend/middleware/webhookSignature.js`
```javascript
const crypto = require('crypto');

const verifyWhatsAppSignature = (req, res, next) => {
  const signature = req.headers['x-hub-signature-256'];
  if (!signature) return res.status(401).send('Missing signature');

  const body = JSON.stringify(req.body);
  const expected = crypto
    .createHmac('sha256', process.env.WHATSAPP_APP_SECRET)
    .update(body)
    .digest('hex');

  if (signature !== `sha256=${expected}`) {
    return res.status(401).send('Invalid signature');
  }
  next();
};

module.exports = { verifyWhatsAppSignature };
```

#### `backend/middleware/tenantCache.js`
```javascript
const Redis = require('ioredis');
const Tenant = require('../models/Tenant');

const redis = new Redis(process.env.REDIS_URL);

const getTenantFromCache = async (subdomain) => {
  const key = `tenant:${subdomain}`;
  const cached = await redis.get(key);
  if (cached) return JSON.parse(cached);

  const tenant = await Tenant.findOne({ subdomain });
  if (tenant) {
    await redis.set(key, JSON.stringify(tenant), 'EX', 3600);
  }
  return tenant;
};

module.exports = { getTenantFromCache };
```

#### `backend/middleware/rateLimiter.js`
```javascript
const Redis = require('ioredis');
const redis = new Redis(process.env.REDIS_URL);

const globalLimiter = async (req, res, next) => {
  const ip = req.ip;
  const key = `rl:global:${ip}`;
  const current = await redis.incr(key);
  if (current === 1) await redis.expire(key, 900);
  if (current > 100) return res.status(429).json({ error: 'Too many requests' });
  next();
};

const tenantLimiter = (maxRequests = 1000, windowMs = 60000) => async (req, res, next) => {
  const tenantId = req.tenant?._id || 'public';
  const ip = req.ip;
  const key = `rl:tenant:${tenantId}:ip:${ip}`;
  const current = await redis.incr(key);
  if (current === 1) await redis.expire(key, Math.ceil(windowMs / 1000));
  if (current > maxRequests) return res.status(429).json({ error: 'Too many requests' });
  next();
};

module.exports = { globalLimiter, tenantLimiter };
```

```javascript
const express = require('express');
const router = express.Router();
const { handleIncomingMessage } = require('../services/webhookHandler');
const { verifyWhatsAppSignature } = require('../middleware/webhookSignature');
const { getTenantFromCache } = require('../middleware/tenantCache');

router.get('/webhook', (req, res) => {
  const mode = req.query['hub.mode'];
  const token = req.query['hub.verify_token'];
  const challenge = req.query['hub.challenge'];

  if (mode === 'subscribe' && token === process.env.WHATSAPP_WEBHOOK_VERIFY_TOKEN) {
    res.status(200).send(challenge);
  } else {
    res.sendStatus(403);
  }
});

router.post('/webhook', verifyWhatsAppSignature, async (req, res) => {
  try {
    const tenantSubdomain = req.headers['x-tenant'] || req.query.tenant;
    if (!tenantSubdomain) return res.status(400).send('Tenant ID missing');

    const tenant = await getTenantFromCache(tenantSubdomain);
    if (!tenant) return res.sendStatus(404);

    await handleIncomingMessage(tenant._id, req.body);
    res.sendStatus(200);
  } catch (error) {
    console.error(error);
    res.sendStatus(500);
  }
});

module.exports = router;
```

```javascript
import React, { createContext, useState, useEffect } from 'react';
import api from '../utils/api';

export const AuthContext = createContext();

export const AuthProvider = ({ children }) => {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const token = localStorage.getItem('token');
    if (token) {
      api.get('/auth/me')
        .then(res => setUser(res.data))
        .catch(() => {
          localStorage.removeItem('token');
          setUser(null);
        })
        .finally(() => setLoading(false));
    } else {
      setLoading(false);
    }
  }, []);

  const login = async (email, password) => {
    const res = await api.post('/auth/login', { email, password });
    localStorage.setItem('token', res.data.token);
    setUser(res.data.user);
    return res.data;
  };

  const logout = () => {
    localStorage.removeItem('token');
    setUser(null);
  };

  return (
    <AuthContext.Provider value={{ user, login, logout, loading }}>
      {children}
    </AuthContext.Provider>
  );
};
```

```javascript
const User = require('../models/User');
const Tenant = require('../models/Tenant');
const bcrypt = require('bcryptjs');
const jwt = require('jsonwebtoken');

exports.register = async (req, res) => {
  try {
    const { name, email, password, tenantSubdomain } = req.body;

    let tenant = await Tenant.findOne({ subdomain: tenantSubdomain });
    if (!tenant) {
      tenant = new Tenant({
        name: tenantSubdomain,
        subdomain: tenantSubdomain,
        branding: { primaryColor: '#25D366', secondaryColor: '#075E54' }
      });
      await tenant.save();
    }

    const hashedPassword = await bcrypt.hash(password, 12);
    const user = new User({ name, email, password: hashedPassword, tenantId: tenant._id });
    await user.save();

    const token = jwt.sign({ userId: user._id }, process.env.JWT_SECRET, { expiresIn: '7d' });
    res.status(201).json({ token, user });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};

exports.login = async (req, res) => {
  try {
    const { email, password } = req.body;
    const user = await User.findOne({ email }).populate('tenantId');
    if (!user || !user.isActive) return res.status(401).json({ error: 'Invalid credentials' });

    const isMatch = await bcrypt.compare(password, user.password);
    if (!isMatch) return res.status(401).json({ error: 'Invalid credentials' });

    user.lastSeen = new Date();
    await user.save();

    const token = jwt.sign({ userId: user._id }, process.env.JWT_SECRET, { expiresIn: '7d' });
    res.json({ token, user });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};

exports.getMe = async (req, res) => {
  res.json(req.user);
};
```

```javascript
const axios = require('axios');

class WhatsAppService {
  constructor(tenant) {
    this.accessToken = tenant.whatsappConfig.accessToken;
    this.phoneNumberId = tenant.whatsappConfig.phoneNumberId;
    this.baseUrl = `https://graph.facebook.com/v18.0/${this.phoneNumberId}`;
  }

  async sendTextMessage(to, text) {
    try {
      const response = await axios.post(
        `${this.baseUrl}/messages`,
        {
          messaging_product: 'whatsapp',
          recipient_type: 'individual',
          to,
          type: 'text',
          text: { body: text }
        },
        {
          headers: {
            Authorization: `Bearer ${this.accessToken}`,
            'Content-Type': 'application/json'
          }
        }
      );
      return response.data;
    } catch (error) {
      console.error('WhatsApp send error:', error.response?.data || error.message);
      throw error;
    }
  }

  async sendTemplateMessage(to, templateName, language = 'fr', components = []) {
    try {
      const response = await axios.post(
        `${this.baseUrl}/messages`,
        {
          messaging_product: 'whatsapp',
          to,
          type: 'template',
          template: {
            name: templateName,
            language: { code: language },
            components
          }
        },
        {
          headers: {
            Authorization: `Bearer ${this.accessToken}`,
            'Content-Type': 'application/json'
          }
        }
      );
      return response.data;
    } catch (error) {
      console.error('WhatsApp send template error:', error.response?.data || error.message);
      throw error;
    }
  }

  async createTemplate(name, category, language, components) {
    try {
      const response = await axios.post(
        `https://graph.facebook.com/v18.0/${this.phoneNumberId}/message_templates`,
        {
          name,
          category,
          language,
          components
        },
        {
          headers: {
            Authorization: `Bearer ${this.accessToken}`,
            'Content-Type': 'application/json'
          }
        }
      );
      return response.data;
    } catch (error) {
      console.error('WhatsApp create template error:', error.response?.data || error.message);
      throw error;
    }
  }

  async sendInteractiveMessage(to, interactiveObject) {
    try {
      const response = await axios.post(
        `${this.baseUrl}/messages`,
        {
          messaging_product: 'whatsapp',
          to,
          type: 'interactive',
          interactive: interactiveObject
        },
        {
          headers: {
            Authorization: `Bearer ${this.accessToken}`,
            'Content-Type': 'application/json'
          }
        }
      );
      return response.data;
    } catch (error) {
      console.error('WhatsApp send interactive error:', error.response?.data || error.message);
      throw error;
    }
  }

  async markAsRead(messageId) {
    try {
      await axios.post(
        `${this.baseUrl}/messages`,
        {
          messaging_product: 'whatsapp',
          status: 'read',
          message_id: messageId
        },
        {
          headers: {
            Authorization: `Bearer ${this.accessToken}`,
            'Content-Type': 'application/json'
```

```javascript
const Contact = require('../models/Contact');

async function handleIncomingMessage(tenantId, payload) {
  const entry = payload.entry[0];
  const changes = entry.changes[0];
  const value = changes.value;
  const messages = value.messages;

  if (!messages) return;

  for (const msg of messages) {
    const from = msg.from;
    const text = msg.text?.body || '';

    let contact = await Contact.findOne({ tenantId, whatsappId: from });
    if (!contact) {
      contact = new Contact({ tenantId, whatsappId: from, phoneNumber: from });
      await contact.save();
    }

    let conversation = await Conversation.findOne({ tenantId, whatsappId: from, status: { $ne: 'resolved' } });
    if (!conversation) {
      conversation = new Conversation({
        tenantId,
        contactId: contact._id,
        whatsappId: from,
        status: 'open'
      });
      await conversation.save();
    }

    const message = new Message({
      conversationId: conversation._id,
      tenantId,
      direction: 'inbound',
      body: text,
      type: 'text',
      status: 'delivered'
    });
    await message.save();

    conversation.lastMessageAt = new Date();
    await conversation.save();

    // Analyse IA
    const aiService = new AIService(process.env.OPENAI_API_KEY);
    const analysis = await aiService.analyzeMessage(text, { contactId: contact._id });
    
    if (analysis && analysis.should_escalate) {
      // Logique d'escalade
    }
  }
}

module.exports = { handleIncomingMessage };
```

```javascript
const OpenAI = require('openai');

class AIService {
  constructor(apiKey) {
    this.client = new OpenAI({ apiKey });
  }

  async analyzeMessage(message, context = {}) {
    const prompt = `
      Analyse le message suivant reçu d'un client WhatsApp :
      "${message}"
      
      Contexte : ${JSON.stringify(context)}
      
      Réponds avec un objet JSON contenant :
      - sentiment: positif, negatif, neutre
      - intent: commande, question, plainte, autre
      - suggested_reply: suggestion de réponse
      - should_escalate: true/false
    `;
    
    try {
      const completion = await this.client.chat.completions.create({
        model: 'gpt-4',
        messages: [{ role: 'user', content: prompt }],
        response_format: { type: 'json_object' }
      });
      
      return JSON.parse(completion.choices[0].message.content);
    } catch (error) {
      console.error('AI analysis error:', error);
      return null;
    }
  }
}

module.exports = AIService;
```

```javascript
class FlowBuilderService {
  constructor() {
    this.flows = [];
  }

  async executeFlow(flowId, payload) {
    const flow = this.flows.find(f => f.id === flowId);
    if (!flow) return null;

    let currentStep = flow.startStep;
    while (currentStep) {
      const result = await this.executeStep(currentStep, payload);
      if (result.nextStepId) {
        currentStep = flow.steps.find(s => s.id === result.nextStepId);
      } else {
        break;
      }
    }
  }

  async executeStep(step, payload) {
    switch (step.type) {
      case 'send_message':
        // Logique d'envoi de message
        return { nextStepId: step.nextStepId };
      case 'wait_for_response':
        // Attendre réponse
        return { nextStepId: step.nextStepId };
      default:
        return {};
    }
  }
}

module.exports = FlowBuilderService;
```

```javascript
const cron = require('node-cron');
const Campaign = require('../models/Campaign');
const Contact = require('../models/Contact');
const WhatsAppService = require('../services/whatsappService');
const logger = require('../utils/logger');

cron.schedule('* * * * *', async () => {
  const now = new Date();
  try {
    const campaigns = await Campaign.find({
      status: 'scheduled',
      schedule: { $lte: now }
    }).populate('tenantId');

    for (const campaign of campaigns) {
      logger.info(`Démarrage de la campagne ${campaign.name} (${campaign._id})`);
      campaign.status = 'sending';
      await campaign.save();

      try {
        const tenant = campaign.tenantId;
        if (!tenant) {
          logger.error(`Tenant introuvable pour la campagne ${campaign._id}`);
          campaign.status = 'failed';
          await campaign.save();
          continue;
        }

        const whatsappService = new WhatsAppService(tenant);
        const contacts = await Contact.find({ _id: { $in: campaign.recipients } });
        for (const contact of contacts) {
          try {
            await whatsappService.sendTextMessage(contact.whatsappId, campaign.message);
            campaign.sentCount++;
          } catch (sendError) {
            logger.error(`Erreur envoi à ${contact.whatsappId}: ${sendError.message}`);
          }
        }

        campaign.status = 'completed';
        await campaign.save();
        logger.info(`Campagne ${campaign.name} terminée. ${campaign.sentCount} messages envoyés.`);
      } catch (err) {
        logger.error(`Erreur lors de l'exécution de la campagne ${campaign._id}: ${err.message}`);
        campaign.status = 'failed';
        await campaign.save();
      }
    }
  } catch (err) {
    logger.error(`Erreur dans campaignScheduler: ${err.message}`);
  }
});

module.exports = cron;
```

```javascript
import React, { useCallback } from 'react';
import ReactFlow, { Controls, Background, useNodesState, useEdgesState, addEdge } from 'reactflow';
import 'reactflow/dist/style.css';

const initialNodes = [
  { id: '1', type: 'input', position: { x: 0, y: 0 }, data: { label: 'Message reçu' } },
  { id: '2', position: { x: 200, y: 0 }, data: { label: 'Analyser' } },
  { id: '3', position: { x: 400, y: 0 }, data: { label: 'Répondre' } },
];

const initialEdges = [
  { id: 'e1-2', source: '1', target: '2' },
  { id: 'e2-3', source: '2', target: '3' },
];

const FlowCanvas = () => {
  const [nodes, setNodes, onNodesChange] = useNodesState(initialNodes);
  const [edges, setEdges, onEdgesChange] = useEdgesState(initialEdges);
  const onConnect = useCallback((params) => setEdges((eds) => addEdge(params, eds)), [setEdges]);

  return (
    <ReactFlow
      nodes={nodes}
      edges={edges}
      onNodesChange={onNodesChange}
      onEdgesChange={onEdgesChange}
      onConnect={onConnect}
    >
      <Background variant="dots" gap={12} size={1} />
      <Controls />
    </ReactFlow>
  );
};
```
